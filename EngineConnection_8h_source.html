139 class SearchStatusWrapper {
140 std::mutex statusLock;
141 SearchStatusCode status =InitialisingSearch;
142 std::condition_variable var;
143 public:
149 void set(SearchStatusCode code) {
150 std::unique_lock<std::mutex> guard(statusLock);
151 this->status = code;
152 var.notify_all();
153 }
154
164 SearchStatusCode swapIfOngoing(SearchStatusCode newStatus) {
165 std::unique_lock<std::mutex> guard(statusLock);
166
167 SearchStatusCode oldStatus = status;
168
169 if (status == OnGoing) {
170 status = newStatus;
171 }
172
173 return oldStatus;
174 }
175
181 SearchStatusCode get() {
182 std::unique_lock<std::mutex> guard(statusLock);
183 return status;
184 }
185
194 SearchStatusCode waitFor(const std::chrono::milliseconds& dur) {
195
196 std::unique_lock<std::mutex> guard(statusLock);
197
198
199 if (status != OnGoing && status != Stopped)
200 return status;
201
202
203 var.wait_for(guard, dur);
204
205 if (status == OnGoing || status == Stopped)
206 status = TimedOut;
207
208 return status;
209 }
210
217 const SearchStatusCode & operator= (const SearchStatusCode& value) {
218 set(value);
219 return value;
220 }
221
222 bool operator == (const SearchStatusCode& value) {
223 return value == get();
224 }
225
226 bool operator != (const SearchStatusCode& value) {
227 return value != get();
228 }

1272 inline void EngineInstance<Move>::quit()
1273 {
1274 std::unique_lock<std::mutex> guard(core->lock);
1275
1276 if (core->quitCommandSend)
1277 return;
1278
1279 core->quitCommandSend = true;
1280 try {
1281 core->sendToEngine("quit\n");
1282 }
1283 catch (PipeClosedException ignored) {
1284 // If we got here, the engine process in in unstable state, but since we are already killing it, we can ignore it.

1288 int counter = 0;
1289
1290 while (core->processWrapper->healthCheck()) {
1291
1292 if (counter == 10) {
1293 core->processWrapper->kill();
1294 break;
1295 }
1296 std::this_thread::sleep_for(std::chrono::milliseconds(100));
1297 ++counter;
