152 SearchStatusCode swapIfOngoing(SearchStatusCode newStatus) {
153 std::unique_lock<std::mutex> guard(statusLock);
154
155 SearchStatusCode oldStatus = status;
156
157 if (status == OnGoing) {
158 status = newStatus;
159 }
160
161 return oldStatus;
162 }

182 SearchStatusCode waitFor(const std::chrono::milliseconds& dur) {
183
184 std::unique_lock<std::mutex> guard(statusLock);
185
186
187 if (status != OnGoing)
188 return status;
189

191 var.wait_for(guard, dur);
192
193 if (status == OnGoing)
194 status = TimedOut;
195
196 return status;
197 }

1071 inline void EngineInstance<Move>::quit()
1072 {
1073 std::unique_lock<std::mutex> guard(core->lock);
1074
1075 if (core->quitCommandSend)
1076 return;
1077
1078 core->quitCommandSend = true;
1079 try {
1080 core->sendToEngine("quit\n");
1081 }
1082 catch (PipeClosedException ignored) {
1083 // If we got here, the engine process in in unstable state, but since we are already killing it, we can ignore it.
1084 };
1085
1086
1087 int counter = 0;
1088
1089 while (core->processWrapper->healthCheck()) {
1090
1091 if (counter == 10) {
1092 core->processWrapper->kill();
1093 break;
1094 }
1095 std::this_thread::sleep_for(std::chrono::milliseconds(100));
1096 ++counter;
1097 };
